function toggleColFilter(icon, colName){
  // 关闭其他下拉
  document.querySelectorAll('.col-filter-dropdown').forEach(function(d){d.remove();});
  document.querySelectorAll('.col-filter-icon').forEach(function(i){i.classList.remove('active');});

  icon.classList.add('active');

  // 如果缓存为空，先拉取全量数据
  if(allRowsCache.length===0){
    icon.textContent='⏳';
    fetch('/api/projects?limit=10000').then(function(r){return r.json();}).then(function(d){
      allRowsCache=d.rows;
      icon.textContent='▾';
      showFilterDropdown(icon,colName);
    }).catch(function(){icon.textContent='▾';icon.classList.remove('active');});
    return;
  }
  showFilterDropdown(icon,colName);
}

function showFilterDropdown(icon,colName){
  var values = [];
  var seen = {};
  allRowsCache.forEach(function(row){
    var v = row[colName];
    if(v && !seen[v]){seen[v]=true;values.push(v);}
  });
  values.sort();

  var selected = colFilters[colName] || [];

  var dd = document.createElement('div');
  dd.className = 'col-filter-dropdown show';
  dd.innerHTML = values.map(function(v){
    var checked = selected.indexOf(v)>=0 ? 'checked' : '';
    return '<label><input type="checkbox" value="'+v.replace(/"/g,'&quot;')+'" '+checked+' onchange="colFilterCheck(this,\''+colName+'\')">'+v+'</label>';
  }).join('') +
  '<div class="filter-actions"><button onclick="clearColFilter(\''+colName+'\')">清除</button><button class="ok" onclick="applyColFilter(\''+colName+'\')">确定</button></div>';

  // 使用 fixed 定位避免被滚动容器截断
  var rect = icon.getBoundingClientRect();
  dd.style.position = 'fixed';
  dd.style.top = (rect.bottom + 2) + 'px';
  dd.style.left = Math.min(rect.left, window.innerWidth - 160) + 'px';
  dd.style.zIndex = '200';
  document.body.appendChild(dd);

  // 点击外部关闭
  setTimeout(function(){
    document.addEventListener('click', function closeDD(e){
      if(!dd.contains(e.target) && e.target !== icon){
        dd.remove();
        icon.classList.remove('active');
        document.removeEventListener('click', closeDD);
      }
    });
  },100);
}

function colFilterCheck(cb, colName){
  if(!colFilters[colName]) colFilters[colName]=[];
  if(cb.checked){
    if(colFilters[colName].indexOf(cb.value)<0) colFilters[colName].push(cb.value);
  } else {
    colFilters[colName]=colFilters[colName].filter(function(v){return v!==cb.value;});
  }
}

  document.querySelectorAll('.col-filter-dropdown').forEach(function(d){d.remove();});

function updateFilterTags(){
  var container = document.getElementById('filterTags');
  var list = document.getElementById('filterTagList');
  var active = Object.keys(colFilters).filter(function(k){return colFilters[k] && colFilters[k].length>0;});
  if(active.length===0){container.style.display='none';return}
  container.style.display='flex';
  list.innerHTML = active.map(function(col){
    var vals = colFilters[col].join('、');
    return '<span style="background:#fdf0f2;color:#c41230;padding:2px 10px;border-radius:2px;font-size:12px;display:inline-flex;align-items:center;gap:6px;border:1px solid #f5c6c3">'+col+'：'+vals+'<span style="cursor:pointer;font-weight:bold" onclick="clearColFilter(\''+col+'\')">×</span></span>';
  }).join('');
}

function applyAllFilters(rows){
  var activeFilters = Object.keys(colFilters).filter(function(k){return colFilters[k] && colFilters[k].length>0;});
  if(activeFilters.length===0) return rows;
  return rows.filter(function(row){
    return activeFilters.every(function(col){
      var vals = colFilters[col];
      return vals.indexOf(row[col]) >= 0;
    });
  });
}

function formatDateValueForInput(value){
  if(!value) return '';
  if(/^\d{4}-\d{2}-\d{2}$/.test(value)) return value;
  var m = value.match(/^(\d{4})\/(\d{1,2})\/(\d{1,2})$/);
  if(m) return `${m[1]}-${m[2].padStart(2,'0')}-${m[3].padStart(2,'0')}`;
  m = value.match(/^(\d{4})年(\d{1,2})月(\d{1,2})日$/);
  if(m) return `${m[1]}-${m[2].padStart(2,'0')}-${m[3].padStart(2,'0')}`;
  return '';
}

function startEdit(td){
  if(td.classList.contains('editing'))return;
  const col=td.dataset.col,name=td.dataset.name;
  const orig=td.textContent==='—'?'':td.textContent;
  td.classList.add('editing');
  const inp=document.createElement('input');
  inp.type = col === '更新日期' ? 'date' : 'text';
  if(col === '更新日期'){
    inp.value = formatDateValueForInput(orig);
    inp.style.width = '130px';
  } else {
    inp.value = orig;
    inp.style.width = (Math.max(orig.length*12,60))+'px';
  }
  td.textContent='';td.appendChild(inp);inp.focus();
  if(col !== '更新日期') inp.select();
  const save=async()=>{
    const v=inp.value;
    td.classList.remove('editing');td.textContent=v||'—';
    try{
      const r=await fetch(`/api/projects/${encodeURIComponent(name)}/cell`,{
        method:'PUT',headers:{'Content-Type':'application/json'},
        body:JSON.stringify({column:col,value:v}),
      });
      if(r.status===401)window.location.href='/login';
      if(r.status===403){showToast('无权限编辑此单元格','error')}
    }catch(e){}
    td.removeEventListener('blur',save,true);
    td.removeEventListener('keydown',onKey,true);
  };
  const onKey=e=>{if(e.key==='Enter')save();if(e.key==='Escape'){td.textContent=orig;td.classList.remove('editing');td.removeEventListener('blur',save,true);td.removeEventListener('keydown',onKey,true)}};
  td.addEventListener('blur',save,true);
  td.addEventListener('keydown',onKey,true);
}

var colResizeInfo=null;
function initColResize(){
  var theadRow=document.querySelector('thead tr');
  if(!theadRow)return;
  var ths=theadRow.querySelectorAll('th');
  ths.forEach(function(th,i){
    // 最后一列（操作列）不加手柄
    if(th.querySelector('.resize-handle'))return;
    var handle=document.createElement('div');
    handle.className='resize-handle';
    handle.addEventListener('mousedown',function(e){
      e.preventDefault();e.stopPropagation();
      colResizeInfo={colIdx:i,startX:e.clientX,startW:th.offsetWidth,th:th};
      th.classList.add('resizing');
      document.body.style.cursor='col-resize';
      document.body.style.userSelect='none';
    });
    th.appendChild(handle);
  });
}
document.addEventListener('mousemove',function(e){
  if(!colResizeInfo)return;
  var dx=e.clientX-colResizeInfo.startX;
  var newW=Math.max(40,colResizeInfo.startW+dx);
  colResizeInfo.th.style.width=newW+'px';
  colResizeInfo.th.style.minWidth=newW+'px';
  colResizeInfo.th.style.maxWidth=newW+'px';
  // 同步更新该列所有td
  var rows=document.querySelectorAll('tbody tr');
  rows.forEach(function(tr){
    var tds=tr.querySelectorAll('td');
    if(tds[colResizeInfo.colIdx]){
      tds[colResizeInfo.colIdx].style.width=newW+'px';
      tds[colResizeInfo.colIdx].style.minWidth=newW+'px';
      tds[colResizeInfo.colIdx].style.maxWidth=newW+'px';
    }
  });
});
document.addEventListener('mouseup',function(){
  if(!colResizeInfo)return;
  colResizeInfo.th.classList.remove('resizing');
  document.body.style.cursor='';
  document.body.style.userSelect='';
  // 保存所有列宽到 localStorage
  saveColWidths();
  colResizeInfo=null;
});

async function showUserModal(){document.getElementById('userModal').classList.add('show');await loadLeaders();loadUsers()}
async function loadLeaders(){
  var r=await fetch('/api/users');var users=await r.json();
  var sel=document.getElementById('newManager');
  sel.innerHTML='<option value="">无上级</option>';
  users.forEach(function(u){
    if(u.role==='leader'||u.role==='admin'){
      sel.innerHTML+=`<option value="${u.username}">${u.display_name||u.username}</option>`;
    }
  });
}
async function loadUsers(){
  var r=await fetch('/api/users');var users=await r.json();
  var tbody=document.querySelector('#userTable tbody');
  tbody.innerHTML=users.map(function(u){
    var roleLabel={'admin':'管理员','leader':'主管','user':'用户'}[u.role]||'用户';
    var managerName=u.manager||'-';
    var resetBtn=`<button class="btn" style="padding:2px 6px;font-size:10px" onclick="resetUserPwd('${u.username}')">重置密码</button>`;
    var delBtn=u.username==='admin'?'':`<button class="btn btn-danger" style="padding:2px 6px;font-size:10px;margin-left:4px" onclick="delUser('${u.username}')">删除</button>`;
    return `<tr><td style="padding:4px 8px;border-bottom:1px solid #f0f0f0">${u.username}</td><td style="padding:4px 8px;border-bottom:1px solid #f0f0f0">${u.display_name||'-'}</td><td style="padding:4px 8px;border-bottom:1px solid #f0f0f0">${roleLabel}</td><td style="padding:4px 8px;border-bottom:1px solid #f0f0f0">${managerName}</td><td style="padding:4px 8px;border-bottom:1px solid #f0f0f0;text-align:center;white-space:nowrap">${resetBtn}${delBtn}</td></tr>`;
  }).join('');
}
async function addUser(){
  var u=document.getElementById('newUsername').value.trim();
  var p=document.getElementById('newPassword').value;
  var n=document.getElementById('newDisplayName').value.trim();
  var rl=document.getElementById('newRole').value;
  var mg=document.getElementById('newManager').value;
  if(!u||!p){showToast('用户名和密码不能为空','error');return}
  var r=await fetch('/api/users',{method:'POST',headers:{'Content-Type':'application/json'},body:JSON.stringify({username:u,password:p,display_name:n,role:rl,manager:mg})});
  var d=await r.json();
  if(d.success){showToast(d.message);document.getElementById('newUsername').value='';document.getElementById('newPassword').value='';document.getElementById('newDisplayName').value='';loadUsers()}
  else showToast(d.message,'error');
}
async function resetUserPwd(u){
  if(!confirm('确认重置用户「'+u+'」的密码为 123456？'))return;
  var r=await fetch('/api/users/'+encodeURIComponent(u)+'/reset_password',{method:'POST',headers:{'Content-Type':'application/json'},body:JSON.stringify({new_password:'123456'})});
  var d=await r.json();
  d.success?showToast('密码已重置为 123456'):showToast(d.message,'error');
}
async function delUser(u){
  if(!confirm('确认删除用户「'+u+'」？'))return;
  var r=await fetch('/api/users/'+encodeURIComponent(u),{method:'DELETE'});var d=await r.json();
  d.success?showToast(d.message):showToast(d.message,'error');
  loadUsers();
}
// 修改密码
function showPwdModal(){
  document.getElementById('pwdModal').classList.add('show');
  document.getElementById('oldPassword').value='';
  document.getElementById('newPassword1').value='';
}
async function changePassword(){
  var oldPwd=document.getElementById('oldPassword').value;
  var newPwd=document.getElementById('newPassword1').value;
  if(!oldPwd||!newPwd){showToast('请填写新旧密码','error');return}
  var r=await fetch('/api/change_password',{method:'POST',headers:{'Content-Type':'application/json'},body:JSON.stringify({old_password:oldPwd,new_password:newPwd})});
  var d=await r.json();
  if(d.success){showToast('密码已修改，下次登录生效');hideModal('pwdModal')}
  else showToast(d.message,'error');
}

async function showBackupModal(){document.getElementById('backupModal').classList.add('show');loadBackups()}
async function loadBackups(){
  var r=await fetch('/api/backup/list');var list=await r.json();
  var tbody=document.querySelector('#backupTable tbody');
  tbody.innerHTML=list.map(function(b){
    var shortName=b.name.length>32?b.name.substring(0,16)+'...'+b.name.slice(-16):b.name;
    var typeTag=b.type==='完整'?'<span style="background:#e8f2fb;color:#1a6fb5;padding:1px 5px;border-radius:3px;font-size:10px">完整</span>':b.type==='总表'?'<span style="background:#fef3c7;color:#b45309;padding:1px 5px;border-radius:3px;font-size:10px">总表</span>':'<span style="background:#f0f0f0;color:#888;padding:1px 5px;border-radius:3px;font-size:10px">自动</span>';
    var restoreBtn=b.name.includes('_恢复前')?'':`<button class="btn" style="padding:2px 5px;font-size:10px;color:#e53935;border-color:#f5c6c3" onclick="restoreBackup('${b.name}')">恢复</button>`;
    return `<tr><td style="padding:4px 6px;border-bottom:1px solid #f0f0f0;font-family:var(--mono);font-size:11px;max-width:260px;overflow:hidden;text-overflow:ellipsis;white-space:nowrap" title="${b.name}">${shortName}</td><td style="padding:4px 6px;border-bottom:1px solid #f0f0f0;text-align:center">${typeTag}</td><td style="padding:4px 6px;border-bottom:1px solid #f0f0f0;text-align:right;font-size:11px">${b.size_kb}KB</td><td style="padding:4px 6px;border-bottom:1px solid #f0f0f0;font-size:11px;white-space:nowrap">${b.time}</td><td style="padding:4px 6px;border-bottom:1px solid #f0f0f0;text-align:center;white-space:nowrap"><a href="/api/backup/download/${b.name}" class="btn" style="padding:2px 5px;font-size:10px">下载</a>${restoreBtn}<button class="btn btn-danger" style="padding:2px 5px;font-size:10px;margin-left:3px" onclick="delBackup('${b.name}')">删除</button></td></tr>`;
  }).join('');
}
async function doBackup(){
  var btype=document.getElementById('backupType').value;
  var r=await fetch('/api/backup',{method:'POST',headers:{'Content-Type':'application/json'},body:JSON.stringify({type:btype})});
  var d=await r.json();
  if(d.success){document.getElementById('backupResult').style.display='block';document.getElementById('backupResult').textContent='✅ 备份成功: '+d.path;loadBackups();showToast('备份完成')}
  else showToast('备份失败','error');
}
async function restoreBackup(fn){
  var isJson=fn.endsWith('.json');
  var warnMsg=isJson?'⚠️ 确认从「'+fn+'」恢复项目数据？\n\n恢复前会自动备份当前数据库，项目数据将按名称匹配合并。':'⚠️ 确认从「'+fn+'」完整恢复数据库？\n\n恢复前会自动备份当前数据，当前全部数据将被覆盖。建议先手动备份。';
  if(!confirm(warnMsg))return;
  var r=await fetch('/api/backup/restore/'+encodeURIComponent(fn),{method:'POST'});
  var d=await r.json();
  if(d.success){showToast('✅ '+d.message);loadBackups();if(!isJson)setTimeout(function(){location.reload()},1500)}
  else showToast(d.message,'error');
}
async function delBackup(fn){
  if(!confirm('确认删除备份「'+fn+'」？'))return;
  await fetch('/api/backup/'+encodeURIComponent(fn),{method:'DELETE'});
  loadBackups();
}

let currentTrackProject='';

async function showTracking(name){
  currentTrackProject=name;
  document.getElementById('trackingTitle').textContent='📋 '+name+' — 跟踪时间轴';
  document.getElementById('trackingModal').classList.add('show');
  // 初始化时间输入框为当前日期
  var now=new Date();var tz=now.getTimezoneOffset();now=new Date(now.getTime()-tz*60000);
  document.getElementById('trackTime').value=now.toISOString().slice(0,10);
  document.getElementById('trackInput').value='';
  await refreshTracking();
}

async function refreshTracking(){
  var r=await fetch('/api/tracking/'+encodeURIComponent(currentTrackProject));var logs=await r.json();
  var tl=document.getElementById('trackingTimeline');
  var empty=document.getElementById('trackingEmpty');
  if(!logs||logs.length===0){tl.style.display='none';empty.style.display='block';return}
  tl.style.display='block';empty.style.display='none';
  tl.innerHTML=logs.map(function(l,i){
    var time=l['操作时间']||'';var content=(l['跟踪情况']||'');
    var operator=l['操作人']||'';var id=l['id'];
    return `<div style="position:relative;padding:10px 0 10px 18px;border-left:2px solid ${i===0?'#6366f1':'#e0e2e6'};margin-left:6px">
      <div style="position:absolute;left:-6px;top:14px;width:10px;height:10px;border-radius:50%;background:${i===0?'#6366f1':'#c8ccd4'};border:2px solid #fff;box-shadow:0 0 0 1px ${i===0?'#6366f1':'#d5d7da'}"></div>
      <div style="display:flex;justify-content:space-between;align-items:flex-start;gap:8px">
        <div style="flex:1;min-width:0">
          <div style="font-size:11px;color:#8f93a0;margin-bottom:3px">${time}  ·  ${operator||'系统'}</div>
          <div style="font-size:13px;color:#2a2f3a;line-height:1.6;white-space:pre-wrap;word-break:break-word">${content}</div>
        </div>
        <button onclick="delTracking(${id})" title="删除此记录" style="flex-shrink:0;width:22px;height:22px;border:none;background:none;color:#c0c4cc;cursor:pointer;font-size:14px;border-radius:3px;line-height:1;padding:0;transition:all .15s" onmouseenter="this.style.color='#e53935';this.style.background='#fef2f2'" onmouseleave="this.style.color='#c0c4cc';this.style.background='none'">✕</button>
      </div>
    </div>`;
  }).join('');
}

async function addTracking(){
  var content=document.getElementById('trackInput').value.trim();
  if(!content){showToast('请输入跟踪内容','error');return}
  var time=document.getElementById('trackTime').value;
  var r=await fetch('/api/tracking/'+encodeURIComponent(currentTrackProject),{
    method:'POST',headers:{'Content-Type':'application/json'},
    body:JSON.stringify({content:content,time:time})
  });
  var d=await r.json();
  if(d.error){showToast(d.error,'error');return}
  document.getElementById('trackInput').value='';
  showToast('✓ 已添加跟踪记录');
  await refreshTracking();
  // 直接更新表格中对应单元格，不重新加载整个表格
  updateTableTrackingCell(currentTrackProject, content, time);
}

async function delTracking(id){
  if(!confirm('确认删除此跟踪记录？'))return;
  await fetch('/api/tracking/'+id,{method:'DELETE'});
  showToast('✓ 已删除');
  await refreshTracking();
  // 删除后也更新表格中对应单元格
  updateTableTrackingCellAfterDel(currentTrackProject);
}
// 直接更新表格中某个项目的跟踪情况单元格
function updateTableTrackingCell(name, content, date){
  var rows = document.querySelectorAll('tbody tr');
  // 找到"跟踪情况"列索引
  var ths = document.querySelectorAll('thead th');
  var colIdx = -1;
  ths.forEach(function(th, i){
    if(th.textContent.trim().indexOf('跟踪情况')===0) colIdx = i;
  });
  if(colIdx < 0) return;
  var displayText = '【' + date + '】' + content;
  // 遍历行找到对应项目
  rows.forEach(function(tr){
    var tds = tr.querySelectorAll('td');
    // 第一列通常是项目名称
    if(tds.length > colIdx && tds[0].textContent.trim() === name){
      var td = tds[colIdx];
      var span = td.querySelector('span');
      if(span){
        span.textContent = displayText.length > 30 ? displayText.substring(0,30)+'...' : displayText;
      }
    }
  });
}

// 删除跟踪记录后更新表格单元格
async function updateTableTrackingCellAfterDel(name){
  try{
    var r = await fetch('/api/tracking/' + encodeURIComponent(name));
    var logs = await r.json();
    if(logs && logs.length > 0){
      var latest = logs[0];
      var time = (latest['操作时间']||'').substring(0,10);
      updateTableTrackingCell(name, latest['跟踪情况']||'', time);
    }
  }catch(e){}
}

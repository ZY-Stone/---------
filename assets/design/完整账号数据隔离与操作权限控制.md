  def _scope_filter(q, user_info):
      role = user_info.get('role')
      username = user_info.get('username')
      dept = user_info.get('dept', '')
      group = user_info.get('group', '')
      if role in ('admin', 'gm', 'operation'):
          return q
      elif role == 'director':
          return q.filter(PotentialCust.dept3 == dept)
      elif role == 'manager':
          return q.filter(PotentialCust.dept5 == group, PotentialCust.dept3 == dept)
      elif role == 'interface':
          return q.filter(PotentialCust.dept3 == dept)
      elif role == 'sales':
          return q.filter(PotentialCust.sales == username)
      return q

【统一权限工具函数】App.hasPerm(perm)
  App.hasPerm = function(perm) {
    var role = (App.loggedInUser || {}).role || 'admin';
    var PERMS = {
      admin:     ['user_manage', 'role_manage', 'product_dict', 'audit_log', 'data_backup', 'data_export', 'data_import', 'product_width', 'potential_product', 'overview'],
      gm:        ['audit_log', 'data_export', 'data_import', 'product_width', 'potential_product', 'overview'],
      operation: ['audit_log', 'data_export', 'data_import', 'product_width', 'potential_product', 'overview'],
      director:  ['data_export', 'product_width', 'potential_product', 'overview'],
      manager:   ['data_export', 'product_width', 'potential_product', 'overview'],
      interface: ['product_width', 'potential_product', 'overview'],
      sales:     ['product_width', 'overview']
    };
    return (PERMS[role] || []).indexOf(perm) >= 0;
  };

【登录后初始化】App.bootstrapPermissions()
  App.bootstrapPermissions = function() {
    var role = App.loggedInUser.role;
    // 隐藏菜单
    if (!App.hasPerm('user_manage')) {
      var userMenu = document.getElementById('menu-user-mgmt');
      if (userMenu) userMenu.style.display = 'none';
    }
    if (!App.hasPerm('role_manage')) {
      var roleMenu = document.getElementById('menu-role-mgmt');
      if (roleMenu) roleMenu.style.display = 'none';
    }
    if (!App.hasPerm('audit_log')) {
      var auditMenu = document.getElementById('menu-audit');
      if (auditMenu) auditMenu.style.display = 'none';
    }
    if (!App.hasPerm('data_backup')) {
      var backupMenu = document.getElementById('menu-backup');
      if (backupMenu) backupMenu.style.display = 'none';
    }
    if (!App.hasPerm('data_import')) {
      var importBtn = document.getElementById('btn-import');
      if (importBtn) importBtn.style.display = 'none';
    }
    if (!App.hasPerm('data_export')) {
      var exportBtn = document.getElementById('btn-export');
      if (exportBtn) exportBtn.style.display = 'none';
    }
    // 锁定顶部筛选（按角色）
    var teamSel = document.querySelector('#page-potential .filter-dept');
    var groupSel = document.querySelector('#page-potential .filter-group-sel');
    var personSel = document.querySelector('#page-potential .filter-person');
    if (role === 'director' && teamSel) {
      teamSel.value = App.loggedInUser.dept;
      teamSel.disabled = true;
    }
    if (role === 'manager') {
      if (teamSel) { teamSel.value = App.loggedInUser.dept; teamSel.disabled = true; }
      if (groupSel) { groupSel.value = App.loggedInUser.group; groupSel.disabled = true; }
    }
    if (role === 'sales') {
      if (teamSel) teamSel.disabled = true;
      if (groupSel) groupSel.disabled = true;
      if (personSel) personSel.value = App.loggedInUser.username;
    }
    if (role === 'interface') {
      if (teamSel) { teamSel.value = App.loggedInUser.dept; teamSel.disabled = true; }
    }
  };

  App.guardRoute = function(pageId) {
    var required = {
      'page-user-mgmt': 'user_manage',
      'page-role-mgmt': 'role_manage',
      'page-audit': 'audit_log',
      'page-backup': 'data_backup',
      'page-dict': 'product_dict'
    };
    var perm = required[pageId];
    if (perm && !App.hasPerm(perm)) {
      App.toast('🚫 无权访问：' + pageId);
      return false;
    }
    return true;
  };

五、后端 API 权限校验（必须）
═══════════════════════════════════════

src/backend/main.py 或 router 装饰器：
  from functools import wraps
  from fastapi import HTTPException, Request

  PERM_MATRIX = {
      'admin': {'*'},
      'gm': {'/api/import/*', '/api/analytics/*', '/api/audit/*'},
      'operation': {'/api/import/*', '/api/analytics/*', '/api/audit/*'},
      'director': {'/api/analytics/*'},
      'manager': {'/api/analytics/*'},
      'interface': {'/api/analytics/*'},
      'sales': {'/api/analytics/overview', '/api/analytics/width/*'}
  }

  def require_perm(perm):
      def decorator(func):
          @wraps(func)
          def wrapper(*args, request: Request, **kwargs):
              user = request.session.get('user') or {}
              role = user.get('role', 'sales')
              allowed = PERM_MATRIX.get(role, set())
              if '*' not in allowed and perm not in allowed:
                  raise HTTPException(403, f'角色 {role} 无权调用 {perm}')
              return func(*args, request=request, **kwargs)
          return wrapper
      return decorator

  def _scope(q, user):
      role, dept, group_, username = user['role'], user['dept'], user['group'], user['username']
      if role in ('admin', 'gm', 'operation'): return q
      if role == 'director': return q.filter(Model.dept3 == dept)
      if role == 'manager': return q.filter(and_(Model.dept5 == group_, Model.dept3 == dept))
      if role == 'interface': return q.filter(Model.dept3 == dept)
      if role == 'sales': return q.filter(Model.sales == username)
      return q
